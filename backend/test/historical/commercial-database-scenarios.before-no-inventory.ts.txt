  async function prepareAvailableLot(options?: {
    installments?: Array<{ amountCents: number; dueOn: string }>;
    paymentMode?: PaymentMode;
  }) {
    const f = await fixture();
    const receptionOperationId = randomUUID();
    const reception = await f.receptionService.create(f.identity, {
      clientOperationId: receptionOperationId,
      driverName: 'Conductor E2E',
      expectedVarietyIds: [f.variety.id],
      providerId: f.provider.id,
      receivedOn: '2026-09-28',
    });
    await f.weighingService.add(f.identity, reception.id, {
      clientOperationId: randomUUID(),
      expectedReceptionVersion: 0,
      grossWeightKg: '50.500',
      tareKg: '0.500',
      varietyId: f.variety.id,
    });
    await f.weighingService.add(f.identity, reception.id, {
      clientOperationId: randomUUID(),
      expectedReceptionVersion: 1,
      grossWeightKg: '49.750',
      tareKg: '0.250',
      varietyId: f.variety.id,
    });
    const closeOperationId = randomUUID();
    await f.weighingService.close(f.identity, reception.id, {
      clientOperationId: closeOperationId,
      expectedReceptionVersion: 2,
    });
    const purchaseOperationId = randomUUID();
    const purchase = await f.purchaseService.settle(f.identity, reception.id, {
      advancesCents: 500,
      clientOperationId: purchaseOperationId,
      discountsCents: 50,
      expectedReceptionVersion: 3,
      freightBasis: CalculationBasis.Sack,
      freightRateCents: 50,
      installments: options?.installments ?? [],
      paymentMode: options?.paymentMode ?? PaymentMode.Cash,
      priceBasis: CalculationBasis.Kilogram,
      unitPriceCents: 100,
    });
    const lot = await f.dataSource
      .getRepository(InventoryLotEntity)
      .findOneByOrFail({
        businessId: f.identity.businessId,
        receptionId: reception.id,
        varietyId: f.variety.id,
      });
    return {
      ...f,
      closeOperationId,
      lot,
      purchase,
      purchaseOperationId,
      reception,
      receptionOperationId,
    };
  }

  it('executes reception, weighing, purchase and cash sale with exact persisted effects', async () => {
    const f = await prepareAvailableLot();
    expect(f.purchase).toMatchObject({
      freightAmountCents: '100',
      grossAmountCents: '9950',
      liquidAmountCents: '9300',
    });
    expect(f.lot).toMatchObject({
      availableSacks: 2,
      availableWeightKg: '99.500',
      initialSacks: 2,
      initialWeightKg: '99.500',
      state: InventoryLotState.Available,
      version: 1,
    });

    const saleOperationId = randomUUID();
    const sale = await f.salesService.create(f.identity, {
      buyerId: f.buyer.id,
      clientOperationId: saleOperationId,
      destinationDepartment: 'Lima',
      discountsCents: 100,
      freightIncludedInTotal: false,
      installments: [],
      lines: [
        {
          expectedLotVersion: 1,
          lotId: f.lot.id,
          priceBasis: CalculationBasis.Kilogram,
          sacks: 1,
          unitPriceCents: 150,
          weightKg: '40.000',
        },
      ],
      paymentMode: PaymentMode.Cash,
    });
    expect(sale).toMatchObject({
      grossAmountCents: '6000',
      totalAmountCents: '5900',
    });
    expect(
      await f.dataSource.getRepository(InventoryLotEntity).findOneByOrFail({
        id: f.lot.id,
      }),
    ).toMatchObject({
      availableSacks: 1,
      availableWeightKg: '59.500',
      state: InventoryLotState.Available,
      version: 2,
    });
    const cash = await f.dataSource
      .getRepository(CashMovementEntity)
      .findBy({ businessId: f.identity.businessId });
    expect(
      cash.map((movement) => ({
        amount: movement.amountCents,
        category: movement.category,
        direction: movement.direction,
      })),
    ).toEqual(
      expect.arrayContaining([
        { amount: '9300', category: 'purchase-payment', direction: 'out' },
        { amount: '5900', category: 'sale-collection', direction: 'in' },
      ]),
    );
    const movements = await f.dataSource
      .getRepository(InventoryMovementEntity)
      .findBy({ businessId: f.identity.businessId, lotId: f.lot.id });
    expect(movements).toHaveLength(2);
    expect(movements.map((movement) => movement.type).sort()).toEqual([
      'purchase_in',
      'sale_out',
    ]);
    expect(
      await f.auditRepository.countBy({ businessId: f.identity.businessId }),
    ).toBe(6);
  });

  it('persists, reuses and invalidates versioned purchase PDF receipts', async () => {
    const f = await prepareAvailableLot();
    const firstPdf = await f.receiptsService.purchase(
      f.identity,
      f.purchase.id,
    );
    const repeatedPdf = await f.receiptsService.purchase(
      f.identity,
      f.purchase.id,
    );
    expect(firstPdf.subarray(0, 5).toString()).toBe('%PDF-');
    expect(repeatedPdf.equals(firstPdf)).toBe(true);

    const firstDocument = await f.receiptDocuments.findOneByOrFail({
      businessId: f.identity.businessId,
      sourceId: f.purchase.id,
      sourceType: 'purchase',
      sourceVersion: `${f.purchase.state}:v${f.purchase.version}`,
    });
    expect(firstDocument).toMatchObject({
      invalidatedAt: null,
      mimeType: 'application/pdf',
      sha256: createHash('sha256').update(firstPdf).digest('hex'),
    });
    expect(firstDocument.retentionUntil.getTime()).toBeGreaterThan(
      Date.now() + 5 * 365 * 24 * 60 * 60_000,
    );
    expect(
      await f.receiptDocuments.countBy({
        businessId: f.identity.businessId,
        sourceId: f.purchase.id,
        sourceType: 'purchase',
      }),
    ).toBe(1);

    const currentReception = await f.receptions.findOneByOrFail({
      id: f.reception.id,
    });
    const voidedPurchase = await f.purchaseService.void(
      f.identity,
      f.purchase.id,
      {
        clientOperationId: randomUUID(),
        expectedPurchaseVersion: f.purchase.version,
        expectedReceptionVersion: currentReception.version,
        reason: 'Anulación E2E para versionar comprobante',
      },
    );
    const voidedPdf = await f.receiptsService.purchase(
      f.identity,
      f.purchase.id,
    );
    expect(voidedPdf.subarray(0, 5).toString()).toBe('%PDF-');
    expect(voidedPurchase.version).toBe(f.purchase.version + 1);

    const documents = await f.receiptDocuments.find({
      order: { createdAt: 'ASC' },
      where: {
        businessId: f.identity.businessId,
        sourceId: f.purchase.id,
        sourceType: 'purchase',
      },
    });
    expect(documents).toHaveLength(2);
    expect(documents[0]?.invalidatedAt).toBeInstanceOf(Date);
    expect(documents[1]).toMatchObject({
      invalidatedAt: null,
      sourceVersion: `voided:v${voidedPurchase.version}`,
    });
    expect(voidedPdf.equals(firstPdf)).toBe(false);

    await expect(
      f.receiptsService.purchase(
        { ...f.identity, businessId: randomUUID() },
        f.purchase.id,
      ),
    ).rejects.toThrow(NotFoundException);
  });

  it('serializes competing sales so only one stale version can commit', async () => {
    const f = await prepareAvailableLot();
    const saleInput = () => ({
      buyerId: f.buyer.id,
      clientOperationId: randomUUID(),
      destinationDepartment: 'Lima',
      discountsCents: 0,
      freightIncludedInTotal: false,
      installments: [],
      lines: [
        {
          expectedLotVersion: 1,
          lotId: f.lot.id,
          priceBasis: CalculationBasis.Kilogram,
          sacks: 1,
          unitPriceCents: 150,
          weightKg: '40.000',
        },
      ],
      paymentMode: PaymentMode.Cash,
    });

    const attempts = await Promise.allSettled([
      f.salesService.create(f.identity, saleInput()),
      f.salesService.create(f.identity, saleInput()),
    ]);
    expect(
      attempts.filter(({ status }) => status === 'fulfilled'),
    ).toHaveLength(1);
    expect(attempts.filter(({ status }) => status === 'rejected')).toHaveLength(
      1,
    );
    expect(await f.sales.countBy({ businessId: f.identity.businessId })).toBe(
      1,
    );
    expect(
      await f.dataSource.getRepository(InventoryLotEntity).findOneByOrFail({
        id: f.lot.id,
      }),
    ).toMatchObject({
      availableSacks: 1,
      availableWeightKg: '59.500',
      version: 2,
    });
    expect(
      await f.dataSource.getRepository(CashMovementEntity).countBy({
        businessId: f.identity.businessId,
        category: 'sale-collection',
      }),
    ).toBe(1);
    expect(
      await f.dataSource.getRepository(InventoryMovementEntity).countBy({
        businessId: f.identity.businessId,
        type: InventoryMovementType.SaleOut,
      }),
